// 以下疑似コード

struct Record {
    u32 tagID;
}

// tagIDが0x81のときのデータ
// 車両の座標・姿勢同期
struct Record_129 : Record {
    u32 vehicleId; // ゲーム中の車両ID
    u32 time;
    u16 bodyCount; // サイズは車両ごとに固定
    Body[] bodies;

    // 車両に含まれるボディの姿勢情報（ヘッダー）
    struct Body {
        u8 type;
    }
    // typeが0x00のときデータ
    // 前回の同期から(相対)姿勢変化がないボディ
    // データ部分は空
    struct Body_00 : Body {
    }
    // typeが0x01のときのデータ
    // 座標をdoubleで同期するボディ
    // 先頭の1件は常にこれ
    struct Body_01 : Body {
        float[4] rotation;
        double[3] position;
    }
    // typeが0x02の時のデータ
    // 座標をfloatで同期するボディ
    // 先頭ボディに対する相対位置・姿勢のようだ
    struct Body_02 : Body {
        float[4] rotation;
        float[3] position;
    }
}

// tagIDが0x39のときのデータ
// 車両の低頻度な位置同期
// クライアントで何に使われているか不明
struct Record_57 : Record { 
    u32 vehicleId;
    double[3] position;
    float azimuth;
}

// tagIDが0x8eのときのデータ
// Popup表示のイベント
// Popup表示コマンドは複数あるが、setPopupScreenのときのもの。
// おそらく構造は変わらない。
struct Record_142 : Record {
    u16 nameSize;
    char[] name;
    u16 show; // setPopupScreenのとき1
    u16 textSize;
    char[] text;
    float x; // screen X
    float y;
    float z; // screen Y
    float renderDistance;
    u32 uiId;
    u32 vehicleParentId;
    u32 objectParentId;
}
